    );
    const snapshot = await getDocs(q);
    
    if (!snapshot.empty) {
      return {
        id: snapshot.docs[0].id,
        data: snapshot.docs[0].data()
      };
    }
  } catch (e) {
    console.log("Method 1 failed:", e);

    if (docSnap.exists()) {
      return {
        id: docSnap.id,
        data: docSnap.data()
      };
    }
  } catch (e) {
    console.log("Method 2 failed:", e);
  }

  // Method 3: Get ALL restaurants and find match
  try {
    const allSnap = await getDocs(collection(db, "restaurants"));
    
    allSnap.forEach(doc => {
      console.log("Restaurant found:", doc.id, doc.data());
      console.log("ownerId in doc:", doc.data().ownerId);
      console.log("current user uid:", user.uid);
      console.log("match:", doc.data().ownerId === user.uid);
    });

    const match = allSnap.docs.find(d =>
